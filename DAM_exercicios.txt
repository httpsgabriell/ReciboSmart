c) Encontre a frequência de um caractere numa String.

fun freqChar() {
    val nome = "gonçalo"
    val letra = 'o'
    var freq = 0
    for (c in nome) {
        if (c == letra) {
           freq++ 
        }
    }
    println ("$freq")
}

g) Calcular o desvio padrão dos dados apresentados num array.

import kotlin.math.pow
import kotlin.math.sqrt

fun main() {
    val dados = doubleArrayOf(10.0, 12.0, 23.0, 23.0, 16.0, 23.0, 21.0, 16.0)
    val media = dados.average()
    val somaQuadrados = dados.sumOf { (it - media).pow(2.0) }    
    val desvioPadrao = sqrt(somaQuadrados / dados.size)

    println("Desvio Padrão: $desvioPadrao")
}

h) Programa para adicionar duas matrizes.

fun main() {
    val matrizA = arrayOf(
        intArrayOf(1, 2, 3),
        intArrayOf(4, 5, 6)
    )
    val matrizB = arrayOf(
        intArrayOf(9, 8, 7),
        intArrayOf(6, 5, 4)
    )
    val soma = Array(matrizA.size) { i ->
        IntArray(matrizA[0].size) { j ->
            matrizA[i][j] + matrizB[i][j]
        }
    }
    for (linha in soma) {
        println(linha.joinToString("\t"))
    }
}
